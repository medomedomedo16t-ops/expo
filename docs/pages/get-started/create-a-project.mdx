async function chargeVodafone(receiverPhone, productId, walletPin) {
  const headers = {
    'User-Agent': 'okhttp/4.11.0',
    'x-agent-operatingsystem': '13',
    'clientId': 'AnaVodafoneAndroid',
    'Accept-Language': 'ar',
    'x-agent-device': 'OPPO CPH2235',
    'x-agent-version': '2024.7.2.1',
    'x-agent-build': '1050',
    'digitalId': '24S0M31T0I9RK',
  };

  // Step 1: seamless token
  const authResp = await fetch(
    'http://mobile.vodafone.com.eg/checkSeamless/realms/vf-realm/protocol/openid-connect/auth?client_id=ana-vodafone-app-seamless',
    { headers }
  );
  const authData = await authResp.json();
  const seamlessToken = authData.seamlessToken;
  const msisdn = authData.msisdn;

  if (!seamlessToken) {
    return { success: false, message: 'تأكد من اتصالك ببيانات فودافون' };
  }

  // Step 2: access token
  const tokenResp = await fetch(
    'https://mobile.vodafone.com.eg/auth/realms/vf-realm/protocol/openid-connect/token',
    {
      method: 'POST',
      headers: {
        ...headers,
        'Accept': 'application/json',
        'silentLogin': 'true',
        'seamlessToken': seamlessToken,
        'firstTimeLogin': 'true',
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: 'grant_type=password&client_secret=b86e30a8-ae29-467a-a71f-65c73f2ff5e3&client_id=cash-app',
    }
  );
  const tokenData = await tokenResp.json();
  const accessToken = tokenData.access_token;

  if (!accessToken) {
    return { success: false, message: 'فشل المصادقة مع فودافون' };
  }

  // Step 3: execute order
  const orderResp = await fetch(
    'https://mobile.vodafone.com.eg/services/dxl/pom/productOrder',
    {
      method: 'POST',
      headers: {
        ...headers,
        'Accept': 'application/json',
        'Content-Type': 'application/json; charset=UTF-8',
        'api-host': 'ProductOrderingManagement',
        'useCase': 'CashFakkaAndMared',
        'api-version': 'v2',
        'msisdn': `0${msisdn}`,
        'Authorization': `Bearer ${accessToken}`,
      },
      body: JSON.stringify({
        channel: { name: 'MobileApp' },
        orderItem: [{
          action: 'insert',
          id: productId,
          product: {
            characteristic: [
              { name: 'PaymentMethod', value: 'VFCash' },
              { name: 'USE_EMONEY', value: 'False' },
              { name: 'MerchantCode', value: '81841829' },
            ],
            id: productId,
            relatedParty: [
              { id: String(msisdn), name: 'MSISDN', role: 'Subscriber' },
              { id: receiverPhone, name: 'Receiver', role: 'Receiver' },
            ],
          },
          '@type': productId,
          eCode: 0,
        }],
        relatedParty: [{ id: walletPin, name: 'pin', role: 'Requestor' }],
        '@type': 'CashFakkaAndMared',
      }),
    }
  );

  const result = await orderResp.json();

  if (result.orderTotalPrice !== undefined || result.state === 'Completed') {
    return { success: true, message: 'تم الشحن بنجاح!' };
  } else {
    return { success: false, message: result.message || 'فشلت العملية' };
  }
}
